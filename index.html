Our Promise

Scientific Piggery & Bio-Security

Every animal at Divya Agro Farms is raised in clean, well-ventilated slatted-floor housing with strictly monitored nutrition and veterinary oversight.

Balanced Grain Nutrition

Our herd is fed a scientifically formulated maize, soy, and mineral-rich ration—never hotel waste—resulting in lean, tender, and flavorful meat.

Strict Bio-Security

Footbaths, sanitized barns, quarantine zones, and regular veterinary vaccination schedules keep our Large White Yorkshire and Duroc herd disease-free.

Step 1: Run This SQL in Supabase

Open Supabase Dashboard → SQL Editor → New Query → Paste & Run

-- 1. Customer Orders Table
create table if not exists public.orders (
  id text primary key,
  created_at timestamptz default now(),
  customer_name text not null,
  phone text not null,
  address text not null,
  slot text not null,
  payment text not null,
  status text default 'Pending',
  items jsonb not null default '[]'::jsonb,
  subtotal numeric not null default 0,
  delivery_fee numeric not null default 0,
  total numeric not null default 0
);

-- 2. Bulk / B2B & Piglet Inquiries Table
create table if not exists public.inquiries (
  ref_no text primary key,
  created_at timestamptz default now(),
  name text not null,
  phone text not null,
  type text not null,
  qty text,
  notes text,
  status text default 'New'
);

-- 3. Daily Product Stock Toggle Table
create table if not exists public.product_stock (
  product_id text primary key,
  in_stock boolean not null default true,
  updated_at timestamptz default now()
);

-- 4. Enable Row Level Security & App Policies
-- (For production, tighten admin update/delete policies using Supabase Auth)
alter table public.orders enable row level security;
alter table public.inquiries enable row level security;
alter table public.product_stock enable row level security;

create policy "Allow storefront orders access" on public.orders
  for all using (true) with check (true);

create policy "Allow storefront inquiries access" on public.inquiries
  for all using (true) with check (true);

create policy "Allow storefront stock access" on public.product_stock
  for all using (true) with check (true);

-- 5. Enable Realtime for instant Admin Dashboard order alerts
alter publication supabase_realtime add table public.orders;
alter publication supabase_realtime add table public.inquiries;
